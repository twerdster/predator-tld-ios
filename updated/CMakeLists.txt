cmake_minimum_required(VERSION 3.20)
project(PredatorRecovered LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PREDATOR_OPENCV_PREFIX "" CACHE PATH "Optional private OpenCV core/imgproc prefix")
if(PREDATOR_OPENCV_PREFIX)
 set(OpenCV_INCLUDE_DIRS "${PREDATOR_OPENCV_PREFIX}/include/opencv4")
 find_library(PREDATOR_CORE NAMES opencv_core PATHS "${PREDATOR_OPENCV_PREFIX}/lib/x86_64-linux-gnu" "${PREDATOR_OPENCV_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
 find_library(PREDATOR_IMGPROC NAMES opencv_imgproc PATHS "${PREDATOR_OPENCV_PREFIX}/lib/x86_64-linux-gnu" "${PREDATOR_OPENCV_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
 set(OpenCV_LIBS ${PREDATOR_CORE} ${PREDATOR_IMGPROC})
else()
 find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc)
endif()
option(PREDATOR_SANITIZE "Instrument recovered project sources with ASan/UBSan" OFF)
if(PREDATOR_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(predator_core src/Tld.cpp src/Tracker.cpp src/TldDetector.cpp src/Classifier.cpp src/Fern.cpp src/PointTest.cpp src/TldImage.cpp src/Utils.cpp)
target_include_directories(predator_core PUBLIC src)
target_include_directories(predator_core SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(predator_core PUBLIC ${OpenCV_LIBS})
add_executable(predator_demo tools/main.cpp)
target_link_libraries(predator_demo PRIVATE predator_core)
enable_testing()
add_test(NAME recovered_core COMMAND predator_demo --self-test)
