cmake_minimum_required(VERSION 3.20)
project(PredatorRecovered LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PREDATOR_OPENCV_PREFIX "" CACHE PATH "Optional private OpenCV core/imgproc prefix")
if(PREDATOR_OPENCV_PREFIX)
 set(OpenCV_INCLUDE_DIRS "${PREDATOR_OPENCV_PREFIX}/include/opencv4")
 find_library(PREDATOR_CORE NAMES opencv_core PATHS "${PREDATOR_OPENCV_PREFIX}/lib/x86_64-linux-gnu" "${PREDATOR_OPENCV_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
 find_library(PREDATOR_IMGPROC NAMES opencv_imgproc PATHS "${PREDATOR_OPENCV_PREFIX}/lib/x86_64-linux-gnu" "${PREDATOR_OPENCV_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
 set(OpenCV_LIBS ${PREDATOR_CORE} ${PREDATOR_IMGPROC})
else()
 find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc)
endif()
option(PREDATOR_SANITIZE "Instrument recovered project sources with ASan/UBSan" OFF)
if(PREDATOR_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(predator_core src/Tld.cpp src/Tracker.cpp src/TldDetector.cpp src/Classifier.cpp src/Fern.cpp src/PointTest.cpp src/TldImage.cpp src/Utils.cpp)
target_include_directories(predator_core PUBLIC src)
target_include_directories(predator_core SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(predator_core PUBLIC ${OpenCV_LIBS})
add_executable(predator_demo tools/main.cpp)
target_link_libraries(predator_demo PRIVATE predator_core)
enable_testing()
add_test(NAME recovered_core COMMAND predator_demo --self-test)

# Same serial Session adapter is linked by UIKit, Android JNI and AppKit frontends.
add_executable(predator_session_check platforms/common/check_session.cpp platforms/common/PredatorSession.cpp)
target_include_directories(predator_session_check PRIVATE platforms/common)
target_link_libraries(predator_session_check PRIVATE predator_core)
add_test(NAME native_session COMMAND predator_session_check ${CMAKE_CURRENT_SOURCE_DIR}/platforms/data/camel-320x180.gray)
set_target_properties(predator_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(predator_session SHARED platforms/common/PredatorCAPI.cpp platforms/common/PredatorSession.cpp)
target_include_directories(predator_session PRIVATE platforms/common)
target_link_libraries(predator_session PRIVATE predator_core)
if(APPLE)
  enable_language(OBJCXX)
  add_executable(PredatorDesktop MACOSX_BUNDLE platforms/macos/main.mm platforms/common/PredatorSession.cpp platforms/data/camel-320x180.gray)
  set_source_files_properties(platforms/data/camel-320x180.gray PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  target_include_directories(PredatorDesktop PRIVATE platforms/common)
  target_link_libraries(PredatorDesktop PRIVATE predator_core "-framework Cocoa")
  target_compile_options(PredatorDesktop PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
  set_target_properties(PredatorDesktop PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.awetzler.predator.desktop MACOSX_BUNDLE_BUNDLE_NAME PredatorDesktop)
  file(GLOB_RECURSE PREDATOR_NOTICES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/platforms/data/licenses/*")
  foreach(notice IN LISTS PREDATOR_NOTICES)
    file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}/platforms/data/licenses" "${notice}")
    get_filename_component(directory "${relative}" DIRECTORY)
    set_source_files_properties("${notice}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/licenses/${directory}")
  endforeach()
  target_sources(PredatorDesktop PRIVATE ${PREDATOR_NOTICES})
endif()
